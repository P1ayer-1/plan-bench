     5	HEADER = ["date", "description", "amount", "category", "tags"]
     6	
     7	
     8	def parse_csv(text):
     9	    """Parse CSV text with header date,description,amount,category,tags.
    10	    tags are separated by ';'. Returns a list of Transaction."""
    11	    lines = [l for l in text.strip().splitlines() if l.strip()]
    12	    if not lines:
    13	        return []
    14	    header = [h.strip() for h in lines[0].split(",")]
    15	    if header != HEADER:
    16	        raise ValueError(f"bad header: {header}")
    17	    out = []
    18	    for line in lines[1:]:
    19	        d, desc, amt, cat, tags = line.split(",")
    20	        out.append(Transaction(
    21	            date=date.fromisoformat(d.strip()),
    22	            description=desc.strip(),
    23	            amount=Money(amt.strip()),
    24	            category=cat.strip(),
    25	            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
    26	        ))
    27	    return out
