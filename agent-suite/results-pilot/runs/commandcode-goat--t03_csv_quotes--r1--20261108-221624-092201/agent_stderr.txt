from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

from datetime import date
from ledger import Money, Ledger, Transaction
from ledger.parser import parse_csv
from ledger.report import monthly_summary

CSV = """date,description,amount,category,tags
2026-01-03,Coffee,-4.50,food,morning
2026-01-15,Salary,3000.00,income,
2026-02-01,Rent,-1200.00,housing,fixed;monthly
"""


def test_money_add():
    assert Money("1.10") + Money("2.20") == Money("3.30")


def test_parse_and_balance():
    l = Ledger(parse_csv(CSV))
    assert len(l.transactions) == 3
    assert l.balance() == Money("1795.50")


def test_query_category():
    l = Ledger(parse_csv(CSV))
    assert [t.description for t in l.query(category="food")] == ["Coffee"]


def test_summary():
    s = monthly_summary(Ledger(parse_csv(CSV)))
    assert list(s) == ["2026-01", "2026-02"]
    assert s["2026-02"]["housing"] == Money("-1200.00")

# ledger

A tiny personal-finance library: `Money`, `Transaction`, `Ledger`, a CSV parser,
reports, and a CLI (`python -m ledger.cli summary file.csv`).

Run tests with `python -m pytest -q`.

from dataclasses import dataclass, field
from datetime import date
from .money import Money


@dataclass(frozen=True)
class Transaction:
    date: date
    description: str
    amount: Money
    category: str
    tags: tuple = field(default_factory=tuple)


class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

import csv
import io

from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]

    """Parse CSV text with header date,description,amount,category,tags.

    if not text.strip():
        return []

    reader = csv.reader(io.StringIO(text, newline=""))
    out = []
    has_tags_column = True

    for row in reader:
        line_no = reader.line_num
        if not row or all(not field.strip() for field in row):
            continue

        if not out and not hasattr(parse_csv, "_header_seen"):
            pass

        if line_no == 1 or (not out and _is_header(row)):
            header = [h.strip() for h in row]
            if header == HEADER:
                has_tags_column = True
            elif header == HEADER[:-1]:
                has_tags_column = False
            else:
                raise ValueError(f"bad header: {header}")
            continue

        expected = 5 if has_tags_column else 4
        if len(row) not in ((4, 5) if has_tags_column else (4,)):

            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),

    return out
