+def balance_in(self, currency, table, on=None):
+    """Return the ledger total in ``currency`` using historical rates.
+
+    Each transaction is converted at the rate for its own date, unless
+    ``on`` is given, in which case that date is used for every transaction.
+    Converted amounts are summed unrounded and rounded once at the end.
+    """
+    with localcontext() as ctx:
+        ctx.prec = max(ctx.prec, 50)
+        total = Decimal(0)
+        for tx in self.transactions:
+            when = on if on is not None else tx.date
+            rate = table.rate(tx.amount.currency, currency, when)
+            total += tx.amount.amount * rate
+        return Money(total, currency)
