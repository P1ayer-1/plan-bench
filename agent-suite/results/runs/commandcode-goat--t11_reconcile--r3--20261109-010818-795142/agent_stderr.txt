from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out
from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

    88	def _match_complete_group(ledger_group, bank_group):
    89	    """Optimal matching when every ledger/bank pair is allowed.
    90	
    91	    Matching all of the smaller side to a subset of the larger side on a line
    92	    is solved by a sequence-alignment dynamic program over sorted dates.
    93	    """
    94	    if not ledger_group or not bank_group:
    95	        return []
    96	
    97	    ledger_items = sorted(ledger_group, key=lambda item: item[1].date)
    98	    bank_items = sorted(bank_group, key=lambda item: item[1].date)

   101	        smaller, larger = ledger_items, bank_items
   102	        ledger_is_smaller = True
   103	    else:
   104	        smaller, larger = bank_items, ledger_items
   105	        ledger_is_smaller = False
   106	
   107	    small_count = len(smaller)

   109	    inf = float("inf")

   111	    previous = [0] * (large_count + 1)
   112	    rows = [previous]
   113	    for small_pos in range(1, small_count + 1):
   114	        small_date = smaller[small_pos - 1][1].date
   115	        current = [inf] * (large_count + 1)
   116	        for large_pos in range(1, large_count + 1):
   117	            current[large_pos] = current[large_pos - 1]
   118	            distance = abs((small_date - larger[large_pos - 1][1].date).days)
   119	            candidate = previous[large_pos - 1] + distance
   120	            if candidate < current[large_pos]:
   121	                current[large_pos] = candidate
   122	        rows.append(current)
   123	        previous = current
   124	
   125	    pairs = []
   126	    small_pos = small_count
   127	    large_pos = large_count
   128	    while small_pos > 0:
   129	        small_date = smaller[small_pos - 1][1].date
   130	        distance = abs(
   131	            (small_date - larger[large_pos - 1][1].date).days
   132	        )
   133	        if rows[small_pos][large_pos] == (
   134	            rows[small_pos - 1][large_pos - 1] + distance
   135	        ):
   136	            small_index = smaller[small_pos - 1][0]
   137	            large_index = larger[large_pos - 1][0]
   138	            if ledger_is_smaller:
   139	                pairs.append((small_index, large_index))
   140	            else:
   141	                pairs.append((large_index, small_index))
   142	            small_pos -= 1
   143	            large_pos -= 1
   144	        else:
   145	            large_pos -= 1
   146	
   147	    return pairs

   165	    )

   197	            len(bank_by_date[bank_date]),
   198	            0,
   199	        )
   200	
   201	    candidate_edges = []
   202	    for ledger_pos, ledger_date in enumerate(ledger_dates):
   203	        ledger_capacity = len(ledger_by_date[ledger_date])
   204	        for bank_pos, bank_date in enumerate(bank_dates):
   205	            distance = abs((ledger_date - bank_date).days)
   206	            if distance <= max_days:

   209	                    ledger_base + ledger_pos,
   210	                    bank_base + bank_pos,
   211	                    capacity,

   220	    ledger_queues = {
   221	        ledger_date: list(indices)

   223	    }
   224	    bank_queues = {
   225	        bank_date: list(indices)

   270

   272	    matched = [
   273	        (ledger_txs[ledger_index], bank_txs[bank_index])
   274	        for ledger_index, bank_index in matched_indices
   275	    ]
   276	    unmatched_ledger = [
   277	        tx for index, tx in enumerate(ledger_txs) if not ledger_matched[index]
   278	    ]
   279	    unmatched_bank = [
   280	        tx for index, tx in enumerate(bank_txs) if not bank_matched[index]
   281	    ]
   282

   284	        matched=matched,
   285	        unmatched_ledger=unmatched_ledger,
   286	        unmatched_bank=unmatched_bank,
   287	    )
