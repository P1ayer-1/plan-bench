from dataclasses import dataclass, field
from datetime import date
from .money import Money


@dataclass(frozen=True)
class Transaction:
    date: date
    description: str
    amount: Money
    category: str
    tags: tuple = field(default_factory=tuple)


class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

    assert len(l.transactions) == 3

# ledger

A tiny personal-finance library: `Money`, `Transaction`, `Ledger`, a CSV parser,
reports, and a CLI (`python -m ledger.cli summary file.csv`).

Run tests with `python -m pytest -q`.
=====CLI=====
import argparse, sys
from .ledger import Ledger
from .parser import parse_csv
from .report import monthly_summary


def main(argv=None):
    ap = argparse.ArgumentParser(prog="ledger")
    sub = ap.add_subparsers(dest="cmd", required=True)
    s = sub.add_parser("summary", help="monthly category summary")
    s.add_argument("file")
    args = ap.parse_args(argv)

    with open(args.file) as f:
        ledger = Ledger(parse_csv(f.read()))
    if args.cmd == "summary":
        for month, cats in monthly_summary(ledger).items():
            print(month)
            for cat, total in sorted(cats.items()):
                print(f"  {cat}: {total}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]
