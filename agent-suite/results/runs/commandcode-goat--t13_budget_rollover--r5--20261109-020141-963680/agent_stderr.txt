from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

from dataclasses import dataclass, field
from datetime import date
from .money import Money


@dataclass(frozen=True)
class Transaction:
    date: date
    description: str
    amount: Money
    category: str
    tags: tuple = field(default_factory=tuple)


class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

from datetime import date
from ledger import Money, Ledger, Transaction
from ledger.parser import parse_csv
from ledger.report import monthly_summary

CSV = """date,description,amount,category,tags
2026-01-03,Coffee,-4.50,food,morning
2026-01-15,Salary,3000.00,income,
2026-02-01,Rent,-1200.00,housing,fixed;monthly
"""


def test_money_add():
    assert Money("1.10") + Money("2.20") == Money("3.30")


def test_parse_and_balance():
    l = Ledger(parse_csv(CSV))
    assert len(l.transactions) == 3
    assert l.balance() == Money("1795.50")


def test_query_category():
    l = Ledger(parse_csv(CSV))
    assert [t.description for t in l.query(category="food")] == ["Coffee"]


def test_summary():
    s = monthly_summary(Ledger(parse_csv(CSV)))
    assert list(s) == ["2026-01", "2026-02"]
    assert s["2026-02"]["housing"] == Money("-1200.00")

from decimal import Decimal

from .money import Money


def _parse_month(value):
    if (
        not isinstance(value, str)
        or len(value) != 7
        or value[4] != "-"
        or not value[:4].isdigit()
        or not value[5:].isdigit()
    ):
        raise ValueError(f"invalid month: {value!r}")
    month = int(value[5:])
    if month < 1 or month > 12:
        raise ValueError(f"invalid month: {value!r}")
    return int(value[:4]), month


def _month_index(year, month):
    return year * 12 + month - 1


def _month_from_index(index):
    year, month = divmod(index, 12)
    return f"{year:04d}-{month + 1:02d}"


def budget_report(ledger, budgets, start_month, end_month, rollover_cap=None):
    """Return a per-month budget report with optional rollover."""
    start_year, start_num = _parse_month(start_month)
    end_year, end_num = _parse_month(end_month)
    start = _month_index(start_year, start_num)
    end = _month_index(end_year, end_num)
    if start > end:
        raise ValueError("start_month must not be after end_month")

    spent = {}
    for tx in ledger.transactions:
        index = _month_index(tx.date.year, tx.date.month)
        if index < start or index > end:
            continue
        if tx.category not in budgets:
            continue
        budget = budgets[tx.category]
        if tx.amount.currency != budget.currency:
            raise ValueError(
                f"currency mismatch for category {tx.category!r}: "
                f"budget is {budget.currency}, transaction is {tx.amount.currency}"
            )
        key = (index, tx.category)
        spent[key] = spent.get(key, Decimal(0)) + tx.amount.amount

    result = {}
    carried = {}
    for index in range(start, end + 1):
        month = _month_from_index(index)
        rows = {}
        next_carried = {}
        for category, budget in budgets.items():
            currency = budget.currency
            carried_in = carried.get(category, Money.zero(currency))
            spent_amount = spent.get((index, category), Decimal(0))
            spent_money = Money(-spent_amount, currency)
            available = budget + carried_in
            remaining = available - spent_money
            rows[category] = {
                "budget": budget,
                "carried_in": carried_in,
                "spent": spent_money,
                "available": available,
                "remaining": remaining,
            }

            carry_amount = max(remaining.amount, Decimal(0))
            if rollover_cap is not None:
                carry_amount = min(carry_amount, rollover_cap.amount)
            next_carried[category] = Money(carry_amount, currency)
        carried = next_carried
        result[month] = rows
    return result
