from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

    sub = ap.add_subparsers(dest="cmd", required=True)
    s = sub.add_parser("summary", help="monthly category summary")
    s.add_argument("file")
    args = ap.parse_args(argv)

    with open(args.file) as f:
        ledger = Ledger(parse_csv(f.read()))

        for month, cats in monthly_summary(ledger).items():
            print(month)

    return 0

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

                opts[i].append(j)

        nonlocal best
        if i==n:

    rec(0,set(),[])
    return best

"""Bank-statement reconciliation.

Pair ledger transactions with bank transactions when amounts match exactly
(currency included) and dates are at most ``max_days`` apart.  The pairing
maximises the number of pairs and, among those, minimises the total date
distance.
"""

from dataclasses import dataclass, field
from typing import List, Tuple


@dataclass
class Reconciliation:
    matched: List[Tuple[object, object]] = field(default_factory=list)
    unmatched_ledger: List[object] = field(default_factory=list)
    unmatched_bank: List[object] = field(default_factory=list)


def _min_cost_assignment(cost):
    """Solve a rectangular assignment problem (Hungarian algorithm).

    ``cost`` is a list of rows (ledger candidates) each a list of column
    costs (bank candidates); every row must have at least as many columns as
    there are rows.  Returns a list mapping row index -> column index,
    minimising the total cost.  Rows may be assigned any column, but since the
    caller guarantees that entering a row is cheaper than not entering it,
    only optimal assignments are returned.
    """
    n = len(cost)
    if n == 0:
        return []
    m = len(cost[0])
    INF = float("inf")
    u = [0] * (n + 1)
    v = [0] * (m + 1)
    p = [0] * (m + 1)
    way = [0] * (m + 1)
    for i in range(1, n + 1):
        p[0] = i
        j0 = 0
        minv = [INF] * (m + 1)
        used = [False] * (m + 1)
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            row = cost[i0 - 1]
            for j in range(1, m + 1):
                if not used[j]:
                    cur = row[j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(m + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    assignment = [-1] * n
    for j in range(1, m + 1):
        if p[j] != 0:
            assignment[p[j] - 1] = j - 1
    return assignment

def reconcile(ledger_txs, bank_txs, max_days=3):
    """Reconcile ``ledger_txs`` against ``bank_txs``.

    Returns a :class:`Reconciliation` with ``matched`` ordered by the ledger
    transaction's position in ``ledger_txs``.
    """
    ledger_txs = list(ledger_txs)
    bank_txs = list(bank_txs)

    # Bucket bank transactions by amount (Money is hashable, and equality
    # includes the currency).
    bank_by_amount = {}
    for j, btx in enumerate(bank_txs):
        bank_by_amount.setdefault(btx.amount, []).append(j)

    matched_ledger = set()
    matched_bank = set()
    pairs = {}

    for i, ltx in enumerate(ledger_txs):
        candidates = bank_by_amount.get(ltx.amount)
        if not candidates:
            continue
        scored = []
        for j in candidates:
            if j in matched_bank:
                continue
            dist = abs((ltx.date - bank_txs[j].date).days)
            if dist <= max_days:
                scored.append((dist, j))
        if not scored:
            continue
        if len(scored) == 1:
            dist, j = scored[0]
            pairs[i] = j
            matched_ledger.add(i)
            matched_bank.add(j)
            continue
        # Several candidates: solve an assignment problem within this amount
        # group.  Rows are the ledger candidates and columns the bank
        # candidates.  Pad to a square matrix with dummy (> max_days) edges so
        # the optimum never uses a forbidden pair; the padding edges also let
        # a participant stay unmatched, since their cost exceeds any real
        # edge.
        scored.sort()
        bank_cols = [j for _, j in scored]
        ledger_rows = [i]
        for k, other in enumerate(ledger_txs):
            if k == i or k in matched_ledger or other.amount != ltx.amount:
                continue
            for j in bank_cols:
                if j not in matched_bank and abs((other.date - bank_txs[j].date).days) <= max_days:
                    ledger_rows.append(k)
                    break
        p = len(ledger_rows)
        q = len(bank_cols)
        size = max(p, q)
        penalty = (max_days + 1) * (size + 1)
        cost = []
        for k in ledger_rows:
            row = []
            for j in bank_cols:
                if j in matched_bank:
                    row.append(penalty)
                else:
                    dist = abs((ledger_txs[k].date - bank_txs[j].date).days)
                    row.append(dist if dist <= max_days else penalty)
            row.extend([penalty] * (size - q))
            cost.append(row)
        for _ in range(size - p):
            cost.append([penalty] * size)
        assignment = _min_cost_assignment(cost)
        for r, c in enumerate(assignment):
            if r >= p or c >= q or c < 0:
                continue
            lk = ledger_rows[r]
            bj = bank_cols[c]
            dist = abs((ledger_txs[lk].date - bank_txs[bj].date).days)
            if dist <= max_days and lk not in matched_ledger and bj not in matched_bank:
                pairs[lk] = bj
                matched_ledger.add(lk)
                matched_bank.add(bj)

    matched = [(ledger_txs[i], bank_txs[pairs[i]]) for i in range(len(ledger_txs)) if i in pairs]
    unmatched_ledger = [tx for i, tx in enumerate(ledger_txs) if i not in matched_ledger]
    unmatched_bank = [tx for j, tx in enumerate(bank_txs) if j not in matched_bank]
    return Reconciliation(matched=matched, unmatched_ledger=unmatched_ledger, unmatched_bank=unmatched_bank)

def _min_cost_assignment(cost):
    """Solve a rectangular assignment problem (Hungarian algorithm).

    ``cost`` is a list of rows (ledger candidates) each a list of column
    costs (bank candidates); every row must have at least as many columns as
    there are rows.  Returns a list mapping row index -> column index,
    minimising the total cost.  Rows may be assigned any column, but since the
    caller guarantees that entering a row is cheaper than not entering it,
    only optimal assignments are returned.
    """
    n = len(cost)
    if n == 0:
        return []
    m = len(cost[0])
    INF = float("inf")
    u = [0] * (n + 1)
    v = [0] * (m + 1)
    p = [0] * (m + 1)
    way = [0] * (m + 1)
    for i in range(1, n + 1):
        p[0] = i
        j0 = 0
        minv = [INF] * (m + 1)
        used = [False] * (m + 1)
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            row = cost[i0 - 1]
            for j in range(1, m + 1):
                if not used[j]:
                    cur = row[j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(m + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    assignment = [-1] * n
    for j in range(1, m + 1):
        if p[j] != 0:
            assignment[p[j] - 1] = j - 1
    return assignment

import random, datetime
from ledger import Money, Transaction
from ledger.reconcile import reconcile
def brute(L,B,md):
    opts=[[] for _ in L]
    for i,l in enumerate(L):
        for j,b in enumerate(B):
            if l.amount==b.amount and abs((l.date-b.date).days)<=md:
                opts[i].append(j)
    best=[(0,0)]
    def rec(i,used,npairs,dist):
        if i==len(L):
            if npairs>best[0][0] or (npairs==best[0][0] and dist<best[0][1]): best[0]=(npairs,dist)
            return
        rec(i+1,used,npairs,dist)
        for j in opts[i]:
            if j not in used:
                used.add(j); rec(i+1,used,npairs+1,dist+abs((L[i].date-B[j].date).days)); used.remove(j)
    rec(0,set(),0,0)
    return best[0]
random.seed(1)
for t in range(2000):
    base=datetime.date(2026,1,1)
    L=[Transaction(base+datetime.timedelta(random.randint(0,10)),"x",Money(random.choice(["1","2","3","1"])),"c") for _ in range(random.randint(0,6))]

    n,d=brute(L,B,md)

    # used at most once
    assert len({id(b) for _,b in r.matched})==len(r.matched)
