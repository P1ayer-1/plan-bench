import argparse, sys
from .ledger import Ledger
from .parser import parse_csv
from .report import monthly_summary


def main(argv=None):
    ap = argparse.ArgumentParser(prog="ledger")
    sub = ap.add_subparsers(dest="cmd", required=True)
    s = sub.add_parser("summary", help="monthly category summary")
    s.add_argument("file")
    args = ap.parse_args(argv)

    with open(args.file) as f:
        ledger = Ledger(parse_csv(f.read()))
    if args.cmd == "summary":
        for month, cats in monthly_summary(ledger).items():
            print(month)
            for cat, total in sorted(cats.items()):
                print(f"  {cat}: {total}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

from dataclasses import dataclass, field
from datetime import date
from .money import Money


@dataclass(frozen=True)
class Transaction:
    date: date
    description: str
    amount: Money
    category: str
    tags: tuple = field(default_factory=tuple)


class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

from dataclasses import dataclass, field
from collections import deque


@dataclass
class Reconciliation:
    matched: list = field(default_factory=list)
    unmatched_ledger: list = field(default_factory=list)
    unmatched_bank: list = field(default_factory=list)


def _min_cost_max_matching(adjacency):
    """Max-cardinality, min-total-weight matching on a bipartite graph.

    adjacency[left] is a list of (right, weight) pairs. Returns a list of
    (left, right) pairs. Successive shortest augmenting paths over the
    alternating residual graph (SPFA), which yields a maximum-cardinality
    matching of minimum total weight.
    """
    n_left = len(adjacency)
    right_ids = sorted({r for edges in adjacency for r, _ in edges})
    if n_left == 0 or not right_ids:
        return []
    r_index = {r: i for i, r in enumerate(right_ids)}
    n_right = len(right_ids)

    edges = [[(r_index[r], w) for r, w in adj] for adj in adjacency]
    match_l = [-1] * n_left
    match_r = [-1] * n_right
    INF = float("inf")

    while True:
        dist_l = [INF] * n_left
        dist_r = [INF] * n_right
        prev_l = [-1] * n_left
        prev_r = [-1] * n_right
        inq_r = [False] * n_right

        q = deque()
        for i in range(n_left):
            if match_l[i] == -1:
                dist_l[i] = 0

        for i in range(n_left):
            if match_l[i] == -1:
                for j, w in edges[i]:
                    if w < dist_r[j]:
                        dist_r[j] = w
                        prev_r[j] = i
                        if not inq_r[j]:
                            q.append(j)
                            inq_r[j] = True

        while q:
            j = q.popleft()
            inq_r[j] = False
            k = match_r[j]
            if k == -1:
                continue
            for jj, w in edges[k]:
                if jj == j:
                    continue
                nd = dist_r[j] + w
                if nd < dist_r[jj]:
                    dist_r[jj] = nd
                    prev_r[jj] = k
                    if not inq_r[jj]:
                        q.append(jj)
                        inq_r[jj] = True

        best = -1
        bd = INF
        for j in range(n_right):
            if match_r[j] == -1 and dist_r[j] < bd:
                bd = dist_r[j]
                best = j
        if best == -1:
            break

        j = best
        while j != -1:
            i = prev_r[j]
            if i == -1:
                break
            pj = match_l[i]
            match_l[i] = j
            match_r[j] = i
            j = pj

    out = []
    for i in range(n_left):
        if match_l[i] != -1:
            out.append((i, right_ids[match_l[i]]))
    return out


def reconcile(ledger_txs, bank_txs, max_days=3):
    ledger_txs = list(ledger_txs)
    bank_txs = list(bank_txs)

    bank_by_amount = {}
    for b_idx, b in enumerate(bank_txs):
        bank_by_amount.setdefault(b.amount, []).append(b_idx)

    adjacency = []
    for l in ledger_txs:
        candidates = []
        for b_idx in bank_by_amount.get(l.amount, ()):
            b = bank_txs[b_idx]
            delta = abs((l.date - b.date).days)
            if delta <= max_days:
                candidates.append((b_idx, delta))
        adjacency.append(candidates)

    pairs = _min_cost_max_matching(adjacency)

    used_ledger = set()
    used_bank = set()
    matched = []
    for l_idx, b_idx in sorted(pairs):
        used_ledger.add(l_idx)
        used_bank.add(b_idx)
        matched.append((ledger_txs[l_idx], bank_txs[b_idx]))

    unmatched_ledger = [tx for i, tx in enumerate(ledger_txs) if i not in used_ledger]
    unmatched_bank = [tx for i, tx in enumerate(bank_txs) if i not in used_bank]

    return Reconciliation(
        matched=matched,
        unmatched_ledger=unmatched_ledger,
        unmatched_bank=unmatched_bank,
    )
