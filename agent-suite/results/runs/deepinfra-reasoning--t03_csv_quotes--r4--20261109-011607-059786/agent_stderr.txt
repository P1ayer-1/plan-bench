import csv
import io
from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    if not text.strip():
        return []

    reader = csv.reader(io.StringIO(text, newline=""))
    out = []
    header = None
    expected = 5

    for row in reader:
        line_no = reader.line_num
        if not row or (len(row) == 1 and not row[0].strip()):
            continue

        if header is None:
            header = [h.strip() for h in row]
            if header == HEADER:
                expected = 5
            elif header == HEADER[:-1]:
                expected = 4
            else:
                raise ValueError(f"bad header: {header}")
            continue

        if expected == 5:
            if len(row) == 4:
                row = row + [""]
            elif len(row) != 5:
                raise ValueError(
                    f"wrong number of fields on line {line_no}: "
                    f"expected 5, got {len(row)}"
                )
        else:
            if len(row) != 4:
                raise ValueError(
                    f"wrong number of fields on line {line_no}: "
                    f"expected 4, got {len(row)}"
                )
            row = row + [""]

        d, desc, amt, cat, tags = row
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out
