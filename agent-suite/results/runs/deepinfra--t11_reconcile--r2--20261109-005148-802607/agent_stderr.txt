from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

def _solve_dense(ledger_idx, bank_idx, ledger_txs, bank_txs, max_days):
    """Optimal pairing for a single amount group.

    Maximise the number of pairs; among those, minimise total distance.
    Both constraints are encoded in a single cost: every eligible pair costs
    ``distance + penalty`` where ``penalty`` exceeds the maximum possible total
    distance, so adding a pair is always preferable, and within a fixed pair
    count the distance term decides.  Pairs beyond ``max_days`` are forbidden
    with a very large cost.  Padding (unmatched) entries share the same
    zero-cost row/column so they never influence the result.
    """
    n_l = len(ledger_idx)
    n_b = len(bank_idx)
    if n_l == 0 or n_b == 0:
        return []
    max_dist = 0
    for i in ledger_idx:
        di = ledger_txs[i].date
        for j in bank_idx:
            d = abs((di - bank_txs[j].date).days)
            if d > max_dist:
                max_dist = d
    penalty = max_dist * min(n_l, n_b) + 1
    forbidden = penalty * (min(n_l, n_b) + 1)
    n = n_l + n_b
    cost = [[0] * n for _ in range(n)]
    for r, i in enumerate(ledger_idx):
        di = ledger_txs[i].date
        for c, j in enumerate(bank_idx):
            d = abs((di - bank_txs[j].date).days)
            cost[r][c] = d + penalty if d <= max_days else forbidden
    p = _min_cost_assignment(cost, n, n)
    pairs = []
    for r in range(n_l):
        c = p[r]
        if 0 <= c < n_b:
            i = ledger_idx[r]
            j = bank_idx[c]
            if abs((ledger_txs[i].date - bank_txs[j].date).days) <= max_days:
                pairs.append((i, j))
    return pairs

def _min_cost_assignment(cost, n_rows, n_cols):
    """Min-cost assignment on a rectangular matrix via successive shortest
    paths (Hungarian algorithm) in the form used by e-maxx.

    Returns a list `p` of length n_rows (n_rows <= n_cols) where `p[i]` is the
    column assigned to row i.
    """
    INF = float("inf")
    u = [0] * (n_rows + 1)
    v = [0] * (n_cols + 1)
    p = [0] * (n_cols + 1)
    way = [0] * (n_cols + 1)
    for i in range(1, n_rows + 1):
        p[0] = i
        j0 = 0
        minv = [INF] * (n_cols + 1)
        used = [False] * (n_cols + 1)
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            row = cost[i0 - 1]
            for j in range(1, n_cols + 1):
                if not used[j]:
                    cur = row[j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(n_cols + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    result = [-1] * n_rows
    for j in range(1, n_cols + 1):
        if p[j] != 0:
            result[p[j] - 1] = j - 1
    return result

def _min_cost_assignment(cost, n_rows, n_cols):
    """Min-cost assignment on a rectangular matrix via successive shortest
    paths (Hungarian algorithm) in the form used by e-maxx.

    Returns a list `p` of length n_rows (n_rows <= n_cols) where `p[i]` is the
    column assigned to row i.
    """
    INF = float("inf")
    u = [0] * (n_rows + 1)
    v = [0] * (n_cols + 1)
    p = [0] * (n_cols + 1)
    way = [0] * (n_cols + 1)
    for i in range(1, n_rows + 1):
        p[0] = i
        j0 = 0
        minv = [INF] * (n_cols + 1)
        used = [False] * (n_cols + 1)
        while True:
            used[j0] = True
            i0 = p[j0]
            delta = INF
            j1 = 0
            row = cost[i0 - 1]
            for j in range(1, n_cols + 1):
                if not used[j]:
                    cur = row[j - 1] - u[i0] - v[j]
                    if cur < minv[j]:
                        minv[j] = cur
                        way[j] = j0
                    if minv[j] < delta:
                        delta = minv[j]
                        j1 = j
            for j in range(n_cols + 1):
                if used[j]:
                    u[p[j]] += delta
                    v[j] -= delta
                else:
                    minv[j] -= delta
            j0 = j1
            if p[j0] == 0:
                break
        while True:
            j1 = way[j0]
            p[j0] = p[j1]
            j0 = j1
            if j0 == 0:
                break
    result = [-1] * n_rows
    for j in range(1, n_cols + 1):
        if p[j] != 0:
            result[p[j] - 1] = j - 1
    return result
