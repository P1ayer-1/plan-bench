from decimal import Decimal, ROUND_HALF_UP

CENT = Decimal("0.01")


class Money:
    """An amount of money in a single currency, stored to the cent."""

    def __init__(self, amount, currency="USD"):
        self.amount = Decimal(str(amount)).quantize(CENT, rounding=ROUND_HALF_UP)
        self.currency = currency

    def __add__(self, other):
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, factor):
        return Money(self.amount * Decimal(str(factor)), self.currency)

    def __neg__(self):
        return Money(-self.amount, self.currency)

    def __eq__(self, other):
        return isinstance(other, Money) and self.amount == other.amount and self.currency == other.currency

    def __hash__(self):
        return hash((self.amount, self.currency))

    def __lt__(self, other):
        return self.amount < other.amount

    def __repr__(self):
        return f"Money('{self.amount}', '{self.currency}')"

    def __str__(self):
        return f"{self.amount} {self.currency}"

    def split(self, n):
        """Split into n parts."""
        part = (self.amount / n).quantize(CENT, rounding=ROUND_HALF_UP)
        return [Money(part, self.currency) for _ in range(n)]

    @classmethod
    def zero(cls, currency="USD"):
        return cls(0, currency)

from dataclasses import dataclass, field
from datetime import date
from .money import Money


@dataclass(frozen=True)
class Transaction:
    date: date
    description: str
    amount: Money
    category: str
    tags: tuple = field(default_factory=tuple)


class Ledger:
    def __init__(self, transactions=None):
        self.transactions = list(transactions or [])

    def add(self, tx):
        self.transactions.append(tx)

    def query(self, category=None, start=None, end=None):
        """Return transactions matching all given filters (dates inclusive)."""
        out = []
        for tx in self.transactions:
            if category is not None and tx.category != category:
                continue
            if start is not None and tx.date < start:
                continue
            if end is not None and tx.date > end:
                continue
            out.append(tx)
        return out

    def balance(self):
        total = Money.zero()
        for tx in self.transactions:
            total = total + tx.amount
        return total

    def dedupe(self):
        """Remove duplicate transactions (same date, description and amount),
        keeping the first occurrence and preserving order."""
        result = []
        for tx in self.transactions:
            key = (tx.date, tx.description, tx.amount)
            if key not in [(t.date, t.description, t.amount) for t in result]:
                result.append(tx)
        self.transactions = result
        return self

from datetime import date
from .money import Money
from .ledger import Transaction

HEADER = ["date", "description", "amount", "category", "tags"]


def parse_csv(text):
    """Parse CSV text with header date,description,amount,category,tags.
    tags are separated by ';'. Returns a list of Transaction."""
    lines = [l for l in text.strip().splitlines() if l.strip()]
    if not lines:
        return []
    header = [h.strip() for h in lines[0].split(",")]
    if header != HEADER:
        raise ValueError(f"bad header: {header}")
    out = []
    for line in lines[1:]:
        d, desc, amt, cat, tags = line.split(",")
        out.append(Transaction(
            date=date.fromisoformat(d.strip()),
            description=desc.strip(),
            amount=Money(amt.strip()),
            category=cat.strip(),
            tags=tuple(t.strip() for t in tags.split(";") if t.strip()),
        ))
    return out

from .money import Money


def monthly_summary(ledger):
    """Return {"YYYY-MM": {category: Money}} with categories summed per month."""
    result = {}
    for tx in ledger.transactions:
        m = f"{tx.date.year:04d}-{tx.date.month:02d}"
        if m not in result:
            result[m] = {}
        if tx.category not in result[m]:
            result[m][tx.category] = Money.zero(tx.amount.currency)
        result[m][tx.category] = result[m][tx.category] + tx.amount
    return dict(sorted(result.items()))


def largest_expenses(ledger, n=3):
    """Return the n most negative transactions, most negative first."""
    expenses = [tx for tx in ledger.transactions if tx.amount.amount < 0]
    return sorted(expenses, key=lambda tx: tx.amount.amount)[:n]

from functools import lru_cache
from ledger import Money, Transaction
from ledger.reconcile import reconcile

def brute(lt, bt, md):
    n,m=len(lt),len(bt)
    @lru_cache(None)
    def f(i, mask):
        if i==n: return (0,0)
        bc, bd = f(i+1, mask)
        bestc,bestd=bc,bd
        for j in range(m):
            if mask>>j & 1: continue
            if lt[i].amount==bt[j].amount:
                d=abs((lt[i].date-bt[j].date).days)
                if d<=md:
                    c,dd=f(i+1, mask|1<<j)
                    cand=(c+1, dd+d)
                    if cand[0]>bestc or (cand[0]==bestc and cand[1]<bestd):
                        bestc,bestd=cand
        return bestc,bestd
    return f(0,0)

from dataclasses import dataclass


@dataclass
class Reconciliation:
    matched: list
    unmatched_ledger: list
    unmatched_bank: list


def _date_distance(a, b):
    return abs((a.date - b.date).days)


def _min_cost_matching(ledger_indices, bank_indices, ledger_txs, bank_txs, max_days):
    """Return a maximum-cardinality, minimum-total-distance matching.

    All transactions passed here have the same amount.  Sorting both sides by
    date lets the problem be solved as an order-preserving matching: any
    crossing pairs can be uncrossed without reducing the number of pairs or
    increasing the total distance.
    """
    n = len(ledger_indices)
    m = len(bank_indices)
    if n == 0 or m == 0:
        return []

    ledger_indices = sorted(ledger_indices, key=lambda i: ledger_txs[i].date)
    bank_indices = sorted(bank_indices, key=lambda i: bank_txs[i].date)

    count = [[0] * (m + 1) for _ in range(n + 1)]
    cost = [[0] * (m + 1) for _ in range(n + 1)]
    parent = [[None] * (m + 1) for _ in range(n + 1)]

    for i in range(1, n + 1):
        tx = ledger_txs[ledger_indices[i - 1]]
        for j in range(1, m + 1):
            best_count = count[i - 1][j]
            best_cost = cost[i - 1][j]
            best_move = "ledger"

            c = count[i][j - 1]
            d = cost[i][j - 1]
            if c > best_count or (c == best_count and d < best_cost):
                best_count, best_cost, best_move = c, d, "bank"

            dist = _date_distance(tx, bank_txs[bank_indices[j - 1]])
            if dist <= max_days:
                c = count[i - 1][j - 1] + 1
                d = cost[i - 1][j - 1] + dist
                if c > best_count or (c == best_count and d < best_cost):
                    best_count, best_cost, best_move = c, d, "pair"

            count[i][j] = best_count
            cost[i][j] = best_cost
            parent[i][j] = best_move

    matched = []
    i, j = n, m
    while i > 0 and j > 0:
        move = parent[i][j]
        if move == "pair":
            matched.append((ledger_indices[i - 1], bank_indices[j - 1]))
            i -= 1
            j -= 1
        elif move == "ledger":
            i -= 1
        else:
            j -= 1

    return matched


def reconcile(ledger_txs, bank_txs, max_days=3):
    """Reconcile ledger transactions against bank transactions.

    Transactions are paired only when their amounts (including currency) are
    equal and their dates differ by at most ``max_days`` days.  The pairing
    maximises the number of pairs, then minimises the total date distance.
    """
    ledger_txs = list(ledger_txs)
    bank_txs = list(bank_txs)

    ledger_by_amount = {}
    for i, tx in enumerate(ledger_txs):
        ledger_by_amount.setdefault(tx.amount, []).append(i)

    bank_by_amount = {}
    for i, tx in enumerate(bank_txs):
        bank_by_amount.setdefault(tx.amount, []).append(i)

    pairs = []
    for amount, ledger_indices in ledger_by_amount.items():
        bank_indices = bank_by_amount.get(amount)
        if not bank_indices:
            continue
        pairs.extend(
            _min_cost_matching(
                ledger_indices, bank_indices, ledger_txs, bank_txs, max_days
            )
        )

    pairs.sort(key=lambda pair: pair[0])

    matched_ledger = {li for li, _ in pairs}
    matched_bank = {bi for _, bi in pairs}

    return Reconciliation(
        matched=[(ledger_txs[li], bank_txs[bi]) for li, bi in pairs],
        unmatched_ledger=[
            tx for i, tx in enumerate(ledger_txs) if i not in matched_ledger
        ],
        unmatched_bank=[
            tx for i, tx in enumerate(bank_txs) if i not in matched_bank
        ],
    )
